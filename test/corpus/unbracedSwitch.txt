==================
Unbraced switch bodies retain one statement
==================

int f(int x) {
  switch (x) case 0: ;
  switch (x) case 0: switch (x) { case 0: goto next; default: return 1; }
  x++;
next:
  switch (x) case 1: return 1;
  switch (x) { case 0: x++; x++; break; default: break; }
  switch (x) case 0: while (x) case 1: x--;
  x++;
  return x;
}
