Attributed case labels remain siblings
================================================================================

void f(int x) {
  switch (x) {
    [[vendor::first]] case 1: {}
    [[vendor::second]] [[vendor::extra]] case 2: break;
    [[vendor::last]] default: break;
  }
}

================================================================================
Attributes on statements inside cases do not move their scope
================================================================================

void f(int x) {
  switch (x) {
    case 1:
      [[vendor::statement]] x++;
      [[fallthrough]];
    [[vendor::label]] case 2:
      switch (x) { [[vendor::nested]] case 3: break; }
      break;
  }
  switch (x) [[vendor::single]] case 4: x++;
  x--;
}
