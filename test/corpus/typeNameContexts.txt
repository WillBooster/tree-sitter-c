================================================================================
Storage classes cannot interleave a cast type name
:error
================================================================================

void f(void) { (int static long)1; }

--------------------------------------------------------------------------------

================================================================================
Storage classes cannot interleave a sizeof type name
:error
================================================================================

void f(void) { sizeof(int static long); }

--------------------------------------------------------------------------------

================================================================================
Trailing storage classes leave following declarations bounded
================================================================================

long static;
int after;
void f(void) { unsigned static; int next; }

--------------------------------------------------------------------------------

      declarator: (identifier)
      parameters: (parameter_list
        (parameter_declaration type: (primitive_type))))
